-- Ensure uuid generation
create extension if not exists pgcrypto;

-- Forms
create table if not exists public.forms (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  slug text unique not null,
  description text,
  is_active boolean default true,
  created_at timestamptz default now(),
  updated_at timestamptz default now()
);

-- Form fields
create table if not exists public.form_fields (
  id uuid primary key default gen_random_uuid(),
  form_id uuid not null references public.forms(id) on delete cascade,
  label text not null,
  field_name text not null,             -- unique per form (UI ensures)
  field_type text not null check (
    field_type in (
      'short_answer','paragraph','radio','checkbox','select','rating','date','time'
    )
  ),
  required boolean default false,
  options text[] default '{}',          -- for radio/checkbox/select
  order_index int default 0,
  created_at timestamptz default now()
);

-- Submissions (store dynamic data as JSONB)
create table if not exists public.form_submissions (
  id uuid primary key default gen_random_uuid(),
  form_id uuid not null references public.forms(id) on delete cascade,
  submitted_at timestamptz default now(),
  data jsonb not null                   -- { field_name: value | [values] }
);

-- updated_at trigger
create or replace function set_updated_at()
returns trigger as $$
begin
  new.updated_at = now();
  return new;
end; $$ language plpgsql;

drop trigger if exists trg_forms_updated_at on public.forms;
create trigger trg_forms_updated_at
before update on public.forms
for each row execute function set_updated_at();

-- Indexes
create index if not exists idx_forms_slug on public.forms(slug);
create index if not exists idx_form_fields_form_id on public.form_fields(form_id);
create index if not exists idx_form_submissions_form_id on public.form_submissions(form_id);

-- RLS
alter table public.forms enable row level security;
alter table public.form_fields enable row level security;
alter table public.form_submissions enable row level security;

-- Public can read active forms and fields (to render public form)
create policy "public_read_active_forms"
on public.forms for select
using (is_active = true);

create policy "public_read_active_form_fields"
on public.form_fields for select
using (exists (select 1 from public.forms f where f.id = form_fields.form_id and f.is_active = true));

-- Public can insert submissions to active forms
create policy "public_insert_submissions"
on public.form_submissions for insert
with check (true);

-- Admin (authenticated) full access
create policy "admin_manage_forms"
on public.forms for all
using (auth.role() = 'authenticated')
with check (auth.role() = 'authenticated');

create policy "admin_manage_form_fields"
on public.form_fields for all
using (auth.role() = 'authenticated')
with check (auth.role() = 'authenticated');

create policy "admin_read_submissions"
on public.form_submissions for select
using (auth.role() = 'authenticated');
